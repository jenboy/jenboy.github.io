class Demo03 {
	public static void main(String[] args) {

		/**
			先声明，再赋值
				先开辟空间，后面再放入数据
		*/
		int a;
		a = 100;

		/**
			声明的同时赋值
				开辟空间的同时放入数据
		*/
		int b = 200;

		/**
			一次声明多个数据
		*/
		int x,y,z;

		/**
			一次声明多个变量并赋值
		*/
		int i=10,j=20,k=30;

		System.out.println("Hello World!");
	}
}

class Demo07 {
	public static void main(String[] args) {
		/**
			整数
			byte	-128~127
			short	-32768~32767
			int		-2147483648~2147483647
			long	
		
		*/

		byte b = 100;
		System.out.println("byte b = " + b);
		b = 110;
		System.out.println("byte b = " + b);
		// b = 130;
		System.out.println("byte b = " + b);

		short s = 32767;
		System.out.println("short s = " + s);

		s = -32768;
		System.out.println("short s = " + s);

		int i = 2147483647;
		System.out.println("int i = " + i);

		i = 2147483647;
		System.out.println("int i = " + i);

		long l;
		l = 2147483647;
		System.out.println("long l = " + l);

		// java语言中所有的整数类型默认是int
		// 在给long类型变量赋值的时候,如果数据过大,需要加上L标记
		l = 21474836470L;
		System.out.println("long l = " + l);
	}
}
