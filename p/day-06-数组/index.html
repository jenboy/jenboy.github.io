public class Demo04 {
	public static void main(String[] args) {
		// 创建数组,存储字符串
		String[] names = {"曹操","曹丕","曹植","曹冲","曹彰"};
		
		// 访问数组元素
		System.out.println(names[0]);
		System.out.println(names[1]);
		System.out.println(names[2]);
		System.out.println(names[3]);
		System.out.println(names[4]);
		
		System.out.println("===================");
		
		// 从索引为0的元素开始
		int index = 0;
		// 遍历到数组长度-1的索引位置
		while (index < names.length) {
			// 访问索引位置的元素
			System.out.println(names[index]);
			// 索引向后移动一位
			index++;
		}
		
		System.out.println("===================");
		for(int i = 0; i < names.length; i++) {
			System.out.println(names[i]);
		}
	}
}

public class Demo02 {
	public static void main(String[] args) {
		/**
		 * 	创建整数数组，存储8个元素
		 * 	1、计算数组中所有元素的和
		 * 	2、计算偶数索引位置元素的和
		 * 	3、计算所有元素的平均值
		 */
		int[] arr = new int[] {4213,5,14,554,542,45,231,666};
		
		// 累加的结果,初始为0
		int sum = 0;
		for (int i = 0; i < arr.length; i++) {
			// 累加每一个元素
			sum = sum + arr[i];
		}
		System.out.println("sum = " + sum);
		System.out.println("平均值" + sum/arr.length);
		System.out.println("平均值" + 1.0*sum/arr.length);
		
		System.out.println("=============================");
		
		// 偶数累加的结果,初始为0
		int sumEven = 0;
		for (int i = 0; i < arr.length; i++) {
			// i是索引,如果i是偶数就累加
			if (i % 2 == 0) {
				sumEven += arr[i];
			}
		}
		System.out.println("偶数索引元素累加的结果:" + sumEven);
	}
}

public class Demo03 {
	public static void main(String[] args) {
        /**
        * System.arraycopy
        * 把源数组中的数组复制到目标数组,可以自定义位置和数量
        * 	src - 源数组。
            srcPos - 源数组中的起始位置。
            dest - 目标数组。
            destPos - 目标数据中的起始位置。
            length - 要复制的数组元素的数量。
        */
        // 存储名字的数组
        String[] nameWei = {"曹操","曹植","曹丕","曹冲","曹彰"};
        String[] nameShu = {"刘禅","赵云","黄忠","张苞","关兴"};
        
        // 把上面二个数组中的所有元素合并到一个数组中
        String[] names = new String[nameWei.length + nameShu.length];
        
        // 把nameWei中的元素复制到names,从nameWei的0位置开始,放入names的0位开始,一共复制nameWei.length个
        /**
        * nameWei 源数组
        * 0 源数组起始位置
        * names 目标数组
        * 0 目标数组的起始位置
        * nameWei.length 复制元素的数量
        */
        System.arraycopy(nameWei, 0, names, 0, nameWei.length);
        
        // 把nameShu中的元素复制到names
        System.arraycopy(nameShu, 0, names, nameWei.length, nameShu.length);
	}
}

			System.out.println(newArr[i]);
		}
	}

}

public class SelectionSort implements IArraySort {

    public int[] sort(int[] sourceArray) throws Exception {
        int[] arr = Arrays.copyOf(sourceArray, sourceArray.length);

        // 总共要经过 N-1 轮比较
        for (int i = 0; i < arr.length - 1; i++) {
            int min = i;

            // 每轮需要比较的次数 N-i
            for (int j = i + 1; j < arr.length; j++) {
                if (arr[j] < arr[min]) {
                    // 记录目前能找到的最小值元素的下标
                    min = j;
                }
            }

            // 将找到的最小值和i位置所在的值进行交换
            if (i != min) {
                int tmp = arr[i];
                arr[i] = arr[min];
                arr[min] = tmp;
            }

        }
        return arr;
    }
}

			 * 需要比较几轮 每次找到一个最大值，需要找arr.length-1次
			 *
			 * 需要比较的内容 每次比较相邻的两个元素
			 */
        // 确定比较的总轮数：数组长度-1
        for (int i = 0; i < arr.length - 1; i++) {
            // 确定每一轮比较的次数
            for (int j = 0; j < arr.length - 1 - i; j++) {
                // 确定比较的内容
                if (arr[j] > arr[j + 1]) {
                    // 前面的元素大于后面的元素,交换它们的位置
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                }
            }
        }
        // 遍历排序之后的数组
        for (int i = 0; i < arr.length; i++) {
            System.out.print(arr[i] + " ");
        }
    }
}

    public static void main(String[] args) {
        int[] arr = new int[] { 35, 285, 1, 565, 6, 32 };
        System.out.println((arr));
        System.out.println(toString(arr));
    }

    /**
		 * 官方的数组转换成字符串的方法
		 * 
		 * @param a
		 * @return
		 */
    public static String toString(int[] a) {
        if (a == null)
            return "null";
        int iMax = a.length - 1;
        if (iMax == -1)
            return "[]";
        StringBuilder b = new StringBuilder();
        b.append('[');
        for (int i = 0;; i++) {
            b.append(a[i]);
            if (i == iMax)
                return b.append(']').toString();
            b.append(", ");
        }
    }
}

		 * 数组转换成字符串
		 * 
		 * @param arr
		 * @return
		 */
		public static String arrToString(int[] arr) {
			// 判断arr是否为空
			if (arr == null) {
				return "null";
			}
			// 判断arr的长度是不是0
			if (arr.length == 0) {
				return "[]";
			}
			String str = "[";
			for (int i = 0; i < arr.length; i++) {
				if (i != arr.length - 1) {
					str = str + arr[i] + ", ";
				} else {
					str = str + arr[i];
				}
			}
			str = str + "]";
			return str;
		}
