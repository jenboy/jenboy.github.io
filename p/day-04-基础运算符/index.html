		如果条件01是false，执行条件02，如果条件02是true，执行代码块02，if结束

	3、执行代码块N

public class Demo03 {
	public static void main(String[] args) {
		/**
		 *	键盘录入星期N，输出这一天的课程

		 *		星期二	语文

		// 判断日期,输出课程
		if (day == 1) {
			System.out.println("星期一,课程有语文数学");
		} else if(day == 2){
			System.out.println("星期二,课程有英语数学");
		} else if(day == 3){
			System.out.println("星期3,课程有英语数学");
		} else if(day == 4){
			System.out.println("星期4,课程有英语数学");
		} else if(day == 5){
			System.out.println("星期5,课程有英语数学");
		} else if(day==6 || day==7){
			System.out.println("星期天");
		} else {
			System.out.println("您输入的时间不对");
		}
		System.out.println("OVER");
	}
}

package com.shine.if01;

import java.util.Scanner;

public class Demo04 {
	public static void main(String[] args) {
		/**
		 * if嵌套
		 * 	乘车检票安检案例
		 * 
		 * 	车票	ticket
		 * 	刀子	knifeLength
		 */
		// 创建键盘录入对象
		Scanner sc = new Scanner(System.in);
		
		// 提示并获取车票
		System.out.println("请出示车票(有票==1,无票==0:");
		int ticket = sc.nextInt();
		
		// 判断是否有车票
		if (ticket > 0) {
			System.out.println("检票通过,请出示刀子长度(mm):");
			// 提示并获取刀子长度
			int knifeLength = sc.nextInt();
			
			// 判断刀子是否合法
			if (knifeLength < 150) {
				System.out.println("安检通过,请排队候车");
			} else {
				System.out.println("你携带了违禁物品,请跟我走一趟");
			}
		} else {
			System.out.println("您还没有购买车票,请先去购票");
		}
	}
}

代码块00
变量 key
switch(key){
	case value01:
		代码块01
		break;
	
	case value02:
		代码块02
		break;
	
	case value03:
		代码块03
		break;
		
	... ...
	default:
		代码块N
		break;
}
代码块N+1

执行流程:
	1、执行代码块00
	2、执行case value01，如果key==value01，执行代码块01，执行break，结束switch
		如果key!=value01，但是key==value02，执行代码块02，执行break，结束switch
			如果key!=value02，key==value03，执行代码块03，执行break，结束switch
				... ...
				如果key和任何value都不相等，执行default，执行代码块N，执行break，结束switch
	3、执行代码块N+1

package com.shine.switch0;

public class Demo01 {
	public static void main(String[] args) {
		/**
		 *	风扇档位和转速
		 */
		
		// 风扇的档位
		int level = 2;
		
		switch (level) {
			case 0:
				// level == 0执行下面的代码
				System.out.println("0档位,风扇处于静止状态");
				break;	// 结束switch
				
			case 1:
				// level == 1执行下面的代码
				System.out.println("1档位,风扇处于微风状态");
				break;	// 结束switch
				
			case 2:
				System.out.println("2档位,风扇处于转速适中");
				break;
	
			case 3:
				System.out.println("3档位,风扇处于快速运行状态");
				break;
				
			default:
				System.out.println("档位不存在...");
				break;
		}
		System.out.println("OVER");
	}
}

执行流程：
	第一轮：

	第二轮:
		i=1; i<5==>true;输出i=1; i++
	第三轮：
		i=2; i<5==>true;输出i=2; i++
	第四轮:
		i=3; i<5==>true;输出i=3; i++
	第五轮:
		i=4; i<5==>true;输出i=4; i++
	第六轮（没有执行）:
		i=5; i<5==>false;	循环结束

package com.shine.while0;

public class Demo02 {
	public static void main(String[] args) {
		/**
		 *	输出练习
		 *		1、输出0~100之间的整数
		 *		2、输出0~100之间的偶数 / 奇数
		 *		3、输出100~0
		 *		4、输出a~z
		 *		5、输出Z~A
		 */
		
		int i = 0;
		while (i <= 100) {
			// 判断i是否偶数
			if (i % 2 == 0) {
				System.out.println(i);
			}
			i++;
		}
		System.out.println("===================");
		
		int j = 0;
		while (j <= 100) {
			System.out.println(j);
			j+=2;
		}
		System.out.println("===================");
		
		// 输出Z~A
		int c = 90;
		while (c >= 65) {
			char cc = (char) c;
			System.out.println(cc);
			c--;
		}
		System.out.println("===================");
		char ccc = 'Z';
		while (ccc>='A') {
			System.out.println(ccc);
			ccc--;
		}
	}
}
