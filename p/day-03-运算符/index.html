  • 也可以是范围大的转型成范围小的

package com.shine.transfer;

public class Demo01 {
	public static void main(String[] args) {
		/**
		 *	类型转换
		 *		向上转型
		 *			小的放入大的
		 *			类型要兼容
		 *			天生安全
		 */
		
		byte b = 100;
		
		// 把一字节的变量 b 放入二字节的 s 空间
		short s = b;
		
		// 把二字节的变量 s 放入字节的 i 空间
		int i = s;
		
		// 把 4 字节的 i 放入 8 字节的 l 空间
		long l = i;
		
		// 数字中的天花板
		double d01 = b;
		double d02 = s;
		double d03 = i;
		double d04 = l;
		
	}
}

package com.shine.transfer;

public class Demo02 {
	public static void main(String[] args) {
		/**
		 *	向下转型
		 *		大的放入小的
		 *		类型兼容
		 *		有一定风险
		 */
		
		long l = 200;
		
		// 把 8 字节的 l 空间放入 4 字节的 i，需要强制转换
		int i = (int) l;
		
		// 把 4 字节的 i 放入 2 字节的 s，需要强制转换
		short s = (short) i;
		
		// 把 2 字节的 s 放入 1 字节的 b，需要强制转换
		byte b = (byte) s;
		
		// 强制转换，如果大空间中的数据没有超出小空间的范围，没有损失
		int ii = 123;
		byte bb = (byte) ii;
		System.out.println(bb);
		
		// 强制转换，如果大空间中的数据超出小空间的范围，有损失，只保留小空间能存储的最大数据
		int iii = 12345;
		byte bbb = (byte) iii;
		System.out.println(bbb);
		
		double dd = 3.14;
		int iiii = (int) dd;
		System.out.println(iiii);
		
	}
}

强制类型转换规则：
整数长度足够，数据完整。
例：int i = 100;	byte b = (byte)i;	//b = 100

整数长度不够，数据截断。
例：int i = 10000;	byte b = (byte)i;	//b = 16（符号位变化，可能变为负数）

小数强转整数，数据截断。
例：double d = 2.5;	int i = (int)d;		//i = 2（小数位舍掉）

字符整数互转，数据完整。
例：char c = 65;	int i = c; 		//i = 65
boolean的取值为true/false，不可与其他类型转换。

package com.shine.operator;

public class Demo01 {

	public static void main(String[] args) {
		/**
		 	+	两个数据相加		/	字符串拼接
			-	两个数据相减
			*	两个数据相乘
			/	两个数据相除
			%	两个数据整除的余数
			++	自增一个数
			--	自减一个数字
		 */
		
		int a = 20;
		int b = 30;
		
		System.out.println("a+b = " + (a+b));
		System.out.println("a-b = " + (a-b));
		System.out.println("a*b = " + (a*b));
		System.out.println("a/b = " + (a/b));	// 整数相除结果还是整数
		System.out.println("a%b = " + (a%b));	// 整除之后的余数
		
		System.out.println(15 % 4);
		System.out.println(18 % 5);
		System.out.println(-18 % 5);
		
		System.out.println(10.0 % 3);
		System.out.println(18.4 % 5);
		System.out.println(1.0 - 0.9);
		
		System.out.println("===============================");
		
		a++;
		System.out.println("a = " + a);
		a++;
		System.out.println("a = " + a);
		a++;
		System.out.println("a = " + a);
		
		++b;
		System.out.println("b = " + b);
		++b;
		System.out.println("b = " + b);
		++b;
		System.out.println("b = " + b);
		
		
		/**
		 * 	++符号表示自增一个数字
		 * 	++符号可以在前，也可以在后
		 * 		在前，变量先自己增加一个数字，再把结果赋值给需要的变量
		 * 		在后，变量先把结果赋值给需要的变量，自己再增加一个数字
		 */
		
		int x = 20;
		int y = 30;
		
		System.out.println(++x);
		System.out.println(x);
		
		System.out.println("=========================");
		
		int result = ++y;
		System.out.println("y = " + y);
		System.out.println("result = " + result);
		
	}

}

package com.shine.operator;

public class Demo02 {
	public static void main(String[] args) {
		/**
		 *	关系运算符
		 *		判断关系是否成立 ==》true / false
		 *	>
		 *	>=
		 *	<
		 *	<=
		 *	==
		 *	!=
		 */
		
		int a = 33;
		int b = 55;
		
		System.out.println("a > b ? " + (a > b));	// false
		System.out.println("a >= b ? " + (a >= b));	// false
		
		System.out.println("a < b ? " + (a < b));	// true
		System.out.println("a <= b ? " + (a <= b));	// true 
		
		System.out.println("a == b ? " + (a == b));	// false
		System.out.println("a != b ? " + (a != b));	// true
		
	}
}

package com.shine.operator;

public class Demo03 {
	public static void main(String[] args) {
		
		int ageA = 20;
		int ageB = 25;
		
		// (你的年龄 > 22) 并且 (他的年龄 > 22)
		System.out.println((ageA > 22) & (ageB > 22));
		
		// (你的年龄 > 22) 或者 (他的年龄 > 22)
		System.out.println((ageA > 22) | (ageB > 22));
		
		// 取反
		System.out.println(!true);
		System.out.println(!false);
		
	}
}

package com.shine.operator;

public class Demo05 {
	public static void main(String[] args) {
		int a = 666;
		int b = 888;
		System.out.println("a大于b吗？" + (a>b));
		
		/**
		 * 	a 大于 b 吗?结果 1: 结果 2;
		 * 		如果 a 大于 b, 表达式结果是结果 1
		 * 		如果 a 不大于 b，表达式结果是结果 2
		 * 	
		 * a > b ? a : b;
		 * a > b ? "a 大于 b" : "a 不大于 b";
		 */
		
		// 获取 a 和 b 中较大的数据
		int result = a > b ? a : b;
		System.out.println(result);
		
		String str = a > b ? "a大" : "a不大";
		System.out.println(str);
	}
}

package com.shine.operator;

public class Demo06 {
	public static void main(String[] args) {
		/**
		 *	BMI 计算公式为：
		 *		BMI = 体重 ÷ 身高 2。（体重单位：千克；身高单位：米。）
		 */
		
		double height = 1.75;
		double weight = 65;
		
		// 计算 BMI
		double BMI = weight / (height*height);
		System.out.println(BMI);
		
		// 使用三元运算符判断是否正常
		String result = (BMI>=20 && BMI<=25) ? "BMI指数正常" : "BMI指数不正常";
		System.out.println(result);
	}
}

package com.shine.input;

// 导入需要使用的工具
import java.util.Scanner;

public class Demo01 {
	public static void main(String[] args) {
		// 创建 Scanner 类型的对象
		Scanner sc = new Scanner(System.in);
		
		// 提示用户输入数据
		System.out.println("请输入你的名字:");
		// 扫描获取用户输入的数据
		String name = sc.next();
		System.out.println("您输入的名字是:" + name);
		
		System.out.println("请输入你的年龄:");
		int age = sc.nextInt();
		System.out.println("您的年龄：" + age);
		
		System.out.println("请输入您的身高:");
		double height = sc.nextDouble();
		System.out.println("您的身高:" + height);
		
		System.out.println("你是否毕业(true/false):");
		boolean isGraduate = sc.nextBoolean();
		System.out.println("您的毕业状态:" + isGraduate);
		
	}
}

package com.shine.input;
// 1、导入 Scanner 工具
import java.util.Scanner;

public class Demo02 {
	public static void main(String[] args) {
		/**
		 * Scanner 使用流程
		 * 	1、导入 Scanner 工具
		 * 	2、创建 Scanner 类型的对象
		 * 	3、提示用户输入数据 01
		 * 	4、获取用户输入的数据 01(数据的类型要匹配)
		 * 	5、提示用户输入数据 02
		 * 	6、获取用户输入的数据 02
		 * 	... ...
		 * 	操作获取的数据
		 */
		// 2、创建 Scanner 类型的对象
		Scanner sc = new Scanner(System.in);
		
		// 3、提示用户输入数据 01
		System.out.println("请输入您的身高（M）:");
		// 4、获取用户输入的数据 01(数据的类型要匹配)
		double height = sc.nextDouble();
		
		// 5、提示用户输入数据 02
		System.out.println("请输入您的体重（KG）");
		// 6、获取用户输入的数据 02
		double weight = sc.nextDouble();
		
		// 操作获取的数据, 计算 BMI
		double BMI = weight / (height*height);
		System.out.println(BMI);
		
		// 使用三元运算符判断是否正常
		String result = (BMI>=20 && BMI<=25) ? "BMI指数正常" : "BMI指数不正常";
		System.out.println(result);
		
	}
}
