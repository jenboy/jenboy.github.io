package com.shine.method;

public class Demo02 {
	public static void main(String[] args) {
		/**
		 * 	函数
		 */
		System.out.println("唧唧复唧唧，木兰当户织。");
		print99();
		System.out.println("不闻机杼声，惟闻女叹息。");
		print99();
		System.out.println("问女何所思，问女何所忆。");
		print99();
		
	}
	
	/**
	 *	打印乘法表的方法
	 */
	public static void print99() {
		for (int i = 1; i <= 9; i++) {
			for (int j = 1; j <= i; j++) {
				System.out.print(j + "*" + i + "=" + i*j + "\t");
			}
			System.out.println();
		}
	}
	
}

概述

  • 函数运行过程中参与运算的数据称为参数
  • 根据位置可以分为形参和实参

方法的形参

  • 形参的位置在方法的声明部分
  • 就是在小括号中声明几个变量，不用赋值

方法的实参

  • 调用方法的时候传入的数据，相当于给形参赋值

package com.shine.method02;

public class Demo02 {
	public static void main(String[] args) {
		double perimeter = getPerimeterOfRectangle(4, 5);
		System.out.println("矩形的周长是:" + perimeter);
		
		boolean ret01 = judgeTriangle(3, 4, 5);
		System.out.println("判断的结果:" + ret01);
				
		boolean ret02 = judgeTriangle(3, 3, 9);
		System.out.println("判断的结果:" + ret02);
		
	}
	
	/**
	 *	计算矩形的周长
	 * @param a	边长a
	 * @param b	边长b
	 * @return	周长
	 */
	public static double getPerimeterOfRectangle(double a,double b) {
		double perimeter = (a+b)*2;
		return perimeter;
	}
	
	// 判断能否构成三角形
	public static boolean judgeTriangle(double a,double b,double c) {
		// 构成三角形条件:任意两条边长之和大于第三条边长
		if ((a+b>c) && (a+c>b) && (b+c>a)) {
			return true;
		}
		return false;
	}
	
}

package com.shine.method;

public class Demo01 {
	public static void main(String[] args) {
		/**
		 * return的用法
		 */
		
		System.out.println("main方法开始...");
		
		show();
		
		System.out.println("main方法结束...");
		
	}
	
	public static void show() {
		System.out.println("show方法开始...");
		
		for (int i = 0; i < 10; i++) {
			System.out.println(i);
			if (i == 6) {
				// break;
				// 结束方法
				return;
			}
		}
		
		System.out.println("show方法结束...");
	}
	
}

return返回结果

  • 如果方法声明了返回值类型，需要在方法结束的位置使用return返回和声明的类型相同的数据
    • 比如方法的声明部分标注返回值类型为int，方法需要返回整数
    • 如果方法的声明部分标注返回值类型为String，方法需要返回字符串
    • 如果方法在if分支中返回结果，要确保if和else中都有返回值才行

方法多级调用

概述

  • 方法中能调用其他方法
    • main方法中调用输出的方法
    • main方法中调用读取输入内容的方法
  • 最后被调用的方法最早结束
    • 如果方法A调用了方法B，方法B调用了方法C
    • 方法C最早结束，方法B再结束，方法A最后结束
      • 按照方法的调用链反向结束

package com.shine.method;

public class Demo03 {
	static int count = 0;
	public static void main(String[] args) {
		
		System.out.println("main方法开始...");
		show();
		System.out.println("main方法结束...");
	}
	
	/**
	 *	递归
	 */
	public static void show(){
		count++;
		System.out.println("我是show方法" + count);
		// 如果count是1000,结束后面的调用
		if (count == 1000) {
			return;
		}
		
		// 调用自己
		show();
	}
	
}

package com.shine.method;

public class Demo04 {
	public static void main(String[] args) {
		/**
		 * 	5!
		 * 	5*4!
		 * 	5*4*3!
		 * 	5*4*3*2!
		 * 	5*4*3*2*1!
		 * 	5*4*3*2*1
		 */
		
		int mul = getMul(5);
		System.out.println(mul);
		
		System.out.println(getMul(10));
	}
	
	/**
	 * 	getMul(5)			5!
	 * 	5*getMul(4)			5*4!
	 * 	5*4*getMul(3)		5*4*3!
	 * 	5*4*3*getMul(2) 	5*4*3*2!
	 * 	5*4*3*2*getMul(1)	5*4*3*2*1!
	 * 	5*4*3*2*1
	 */
	
	/**
	 *	计算num的阶乘
	 * @param num
	 * @return
	 */
	public static int getMul(int num) {
		if (num == 1) {
			return 1;
		}
		return num*getMul(num-1);
	}
	
}
